function setupSystem() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var ensureSheet = function(name, headers) { if (!ss.getSheetByName(name)) { var sh = ss.insertSheet(name); sh.appendRow(headers); sh.getRange(1, 1, 1, headers.length).setBackground("#1e3c72").setFontColor("white").setFontWeight("bold"); } }; ensureSheet("Users", ["Username", "Password", "Role"]); ensureSheet("Finances", ["ID", "Date", "Type", "Category", "Amount", "Notes", "Added By"]); ensureSheet("Clients", ["Client_Name"]); ensureSheet("Drivers", ["Partner_Name"]); ensureSheet("Trip_Types", ["Trip_Name", "Client_Price", "Partner_Rate"]); ensureSheet("Transactions", ["Trip_ID", "Date", "Source", "Client", "Driver", "Trip", "Revenue", "Fee", "Commission", "Status", "Notes"]); } function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); var date = data.date || new Date().toISOString().replace('T', ' ').slice(0, 16); // إضافة مشوار جديد if (data.record_type === "trip") { var sheet = ss.getSheetByName("Transactions"); var lastRow = Math.max(sheet.getLastRow(), 1); var tripId = "TRIP-" + ("000" + lastRow).slice(-4); var commissionFormula = "=G" + (lastRow + 1) + "-H" + (lastRow + 1); sheet.appendRow([ tripId, date, "Web App", data.client_name, data.driver_name, data.trip_name, data.revenue, data.driver_fee, commissionFormula, "مكتمل", data.notes, 0, 0, 0, 0, 0 ]); return ContentService.createTextOutput(JSON.stringify({"status": "success", "trip_id": tripId})).setMimeType(ContentService.MimeType.JSON); } // تعديل مشوار موجود بالفعل if (data.record_type === "edit_trip") { var sheet = ss.getSheetByName("Transactions"); var rows = sheet.getDataRange().getValues(); for(var i = 1; i < rows.length; i++) { if(rows[i][0] === data.trip_id) { var commissionFormula = "=G" + (i + 1) + "-H" + (i + 1); sheet.getRange(i + 1, 2, 1, 10).setValues([[ date, "Web App", data.client_name, data.driver_name, data.trip_name, data.revenue, data.driver_fee, commissionFormula, "مكتمل", data.notes ]]); return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Trip not found"})).setMimeType(ContentService.MimeType.JSON); } if (data.record_type === "finance") { var finSheet = ss.getSheetByName("Finances"); var finId = "FIN-" + new Date().getTime().toString().slice(-6); finSheet.appendRow([finId, date, data.fin_type, data.category, data.amount, data.notes, data.added_by]); return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } if (data.record_type === "update_masters") { var updateSheetData = function(sheetName, headers, items) { var sh = ss.getSheetByName(sheetName); if (!sh) { sh = ss.insertSheet(sheetName); } sh.clearContents(); sh.appendRow(headers); if (items && items.length > 0) { items.forEach(function(item) { if (sheetName === "Trip_Types") { sh.appendRow([item.name, item.price || 0, item.driverRate || 0]); } else { sh.appendRow([item]); } }); } }; updateSheetData("Clients", ["Client_Name"], data.clients); updateSheetData("Drivers", ["Partner_Name"], data.drivers); updateSheetData("Trip_Types", ["Trip_Name", "Client_Price", "Partner_Rate"], data.trips); return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } } catch (error) { return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": error.message})).setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var action = e.parameter.action; if (action === "getInitialData") { var users = [], clients = [], drivers = [], trips = [], recentTrips = []; var uSheet = ss.getSheetByName("Users"); if (uSheet) { var uData = uSheet.getDataRange().getValues(); for (var i = 1; i < uData.length; i++) if (uData[i][0] !== "") users.push({ username: String(uData[i][0]).trim() }); } var cSheet = ss.getSheetByName("Clients"); if (cSheet) { var cData = cSheet.getDataRange().getValues(); for (var i = 1; i < cData.length; i++) if (cData[i][0]) clients.push(String(cData[i][0]).trim()); } var dSheet = ss.getSheetByName("Drivers"); if (dSheet) { var dData = dSheet.getDataRange().getValues(); for (var i = 1; i < dData.length; i++) if (dData[i][0]) drivers.push(String(dData[i][0]).trim()); } var tTypeSheet = ss.getSheetByName("Trip_Types"); if (tTypeSheet) { var tTypeData = tTypeSheet.getDataRange().getValues(); for (var i = 1; i < tTypeData.length; i++) { if (tTypeData[i][0]) trips.push({ name: String(tTypeData[i][0]).trim(), price: parseFloat(tTypeData[i][1]) || 0, driverRate: parseFloat(tTypeData[i][2]) || 0 }); } } var transSheet = ss.getSheetByName("Transactions"); if (transSheet) { var tData = transSheet.getDataRange().getValues(); var start = Math.max(1, tData.length - 15); // سحب آخر 15 مشوار for (var k = tData.length - 1; k >= start; k--) { var rDate = tData[k][1]; var dateStr = (rDate instanceof Date) ? rDate.toISOString().replace('T', ' ').slice(0, 16) : String(rDate); recentTrips.push({ id: String(tData[k][0]), date: dateStr, client: String(tData[k][3]), driver: String(tData[k][4]), trip: String(tData[k][5]), revenue: parseFloat(tData[k][6]) || 0, fee: parseFloat(tData[k][7]) || 0, notes: String(tData[k][10]) || "" }); } } return ContentService.createTextOutput(JSON.stringify({ status: "success", users: users, clients: clients, drivers: drivers, trips: trips, recent: recentTrips })).setMimeType(ContentService.MimeType.JSON); } if (action === "verifyAccess" || action === "login") { var userSheet = ss.getSheetByName("Users"); if (!userSheet) return ContentService.createTextOutput(JSON.stringify({status: "error", message: "شيت المستخدمين غير موجود"})).setMimeType(ContentService.MimeType.JSON); var inputUser = String(e.parameter.usr || e.parameter.username || "").trim().toLowerCase(); var inputPass = String(e.parameter.pwd || e.parameter.password || "").trim(); var data = userSheet.getDataRange().getValues(); for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim().toLowerCase() === inputUser && String(data[i][1]).trim() === inputPass) { return ContentService.createTextOutput(JSON.stringify({status: "success", role: String(data[i][2]).trim()})).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({status: "error", message: "كلمة المرور أو اسم المستخدم خطأ"})).setMimeType(ContentService.MimeType.JSON); } if (action === "getReports") { var repType = e.parameter.type; var repTarget = String(e.parameter.target).trim().toLowerCase(); var fromDate = e.parameter.from; var toDate = e.parameter.to; var sheet = ss.getSheetByName("Transactions"); if (!sheet) return ContentService.createTextOutput(JSON.stringify({status: "success", data: []})).setMimeType(ContentService.MimeType.JSON); var data = sheet.getDataRange().getValues(); var results = []; for (var i = 1; i < data.length; i++) { var rawDate = data[i][1]; var rowDateOnly = ""; var displayDate = ""; if (rawDate instanceof Date) { var m = ("0" + (rawDate.getMonth()+1)).slice(-2); var d = ("0" + rawDate.getDate()).slice(-2); rowDateOnly = rawDate.getFullYear() + "-" + m + "-" + d; displayDate = rowDateOnly + " " + ("0" + rawDate.getHours()).slice(-2) + ":" + ("0" + rawDate.getMinutes()).slice(-2); } else { var strDate = String(rawDate); rowDateOnly = strDate.split('T')[0].split(' ')[0]; displayDate = strDate.replace('T', ' ').slice(0, 16); } if (rowDateOnly >= fromDate && rowDateOnly <= toDate) { var cName = String(data[i][3]).trim(); var pName = String(data[i][4]).trim(); var tName = String(data[i][5]).trim(); var rev = parseFloat(data[i][6]) || 0; var fee = parseFloat(data[i][7]) || 0; if (repType === "client" && cName.toLowerCase() === repTarget) { results.push({ date: displayDate, desc: tName + " (Partner: " + pName + ")", amount: rev }); } else if (repType === "driver" && pName.toLowerCase() === repTarget) { results.push({ date: displayDate, desc: tName + " (Client: " + cName + ")", amount: fee }); } else if (repType === "trip_detailed" && tName.toLowerCase() === repTarget) { results.push({ date: displayDate, desc: "Client: " + cName + " | Partner: " + pName, amount: rev }); } else if (repType === "trips") { results.push({ date: displayDate, trip: tName, client: cName, partner: pName, amount: rev }); } else if (repType === "clients_agg") { results.push({ name: cName, amount: rev }); } else if (repType === "partners_agg") { results.push({ name: pName, amount: fee }); } } } if (repType === "clients_agg" || repType === "partners_agg") { var grouped = {}; results.forEach(function(r) { if (!grouped[r.name]) grouped[r.name] = { count: 0, amount: 0 }; grouped[r.name].count += 1; grouped[r.name].amount += r.amount; }); var aggResults = []; for (var key in grouped) { if(key) aggResults.push({ name: key, count: grouped[key].count, amount: grouped[key].amount }); } results = aggResults; } return ContentService.createTextOutput(JSON.stringify({status: "success", data: results})).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({status: "error", message: "أمر غير معروف"})).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({status: "error", message: error.message})).setMimeType(ContentService.MimeType.JSON); } }
